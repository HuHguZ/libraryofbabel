# Online Library, plan 3 of 4: map, fog of war, teleport

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A map of the Library — a vertical cross-section, column right, floor up — as a minimap in the corner and a full map on M, showing the galleries anyone has ever visited (fog of war), the centre, me and the other readers; a click on any gallery or reader takes you there.

**Architecture:** The world is cut into chunks of 32 columns × 32 floors; a chunk's fog is a 1024-bit set. The server marks a bit the first time a pose arrives in a gallery (per run), broadcasts it, and ORs pending bits into Postgres every 2 s; a client asks for the chunks its map shows (`watch`) and keeps them in `onlineStore.chunks`. The map is a 2-D canvas drawn in a rAF loop from mutable data, with pure view maths (`mapView.ts`) over a BigInt centre. Teleport is navigation to the walk URL of a gallery; the director rebuilds the world there.

**Tech Stack:** as plans 1–2; no new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-04-online-library-design.md` — sections 4, 5, 7. Plans 1 and 2 must be done.

## Global Constraints

- No new dependencies. **No git commits or pushes** unless the user asks; ask at the end.
- The map never re-renders React per frame: it draws on a canvas from `onlineStore` and the stage store, in its own rAF loop (the minimap at ≤ 10 fps).
- Offline the map still works: me, the centre, everything in fog (no chunks arrive).
- A `watch` carries at most 64 chunk keys; the map's zoom range keeps the visible chunks below that (8..48 px per gallery).

```ts
/** The map and the fog of war cut the Library into chunks of 32 columns by 32 floors. */
export const CHUNK = 32n;

const floorDiv = (a: bigint, b: bigint) => (a >= 0n ? a / b : -((-a + b - 1n) / b));
const floorMod = (a: bigint, b: bigint) => ((a % b) + b) % b;

export const chunkOf = (cell: Cell): Cell => ({ x: floorDiv(cell.x, CHUNK), floor: floorDiv(cell.floor, CHUNK) });

/** Key of the chunk a gallery is in. */
export const chunkKeyOf = (cell: Cell): string => cellKey(chunkOf(cell));

/** Bit of a gallery in its chunk: column first, then floor. */
export const cellBit = (cell: Cell): number => Number(floorMod(cell.x, CHUNK) + CHUNK * floorMod(cell.floor, CHUNK));

/** The gallery of bit `bit` of a chunk. */
export const bitCell = (chunk: Cell, bit: number): Cell => ({
  x: chunk.x * CHUNK + BigInt(bit % 32),
  floor: chunk.floor * CHUNK + BigInt(Math.floor(bit / 32)),
});
```

```ts
/** One bit per gallery of a chunk: set once anyone has been there. */
export const FOG_BITS = 1024;
const BYTES = FOG_BITS / 8;

export const emptyFog = (): Uint8Array => new Uint8Array(BYTES);
export const hasBit = (bits: Uint8Array, i: number): boolean => (bits[i >> 3] & (1 << (i & 7))) !== 0;

export function setBit(bits: Uint8Array, i: number): void {
  bits[i >> 3] |= 1 << (i & 7);
}

export function fogToBase64(bits: Uint8Array): string {
  let s = "";
  for (const b of bits) s += String.fromCharCode(b);
  return btoa(s);
}

/** Whatever bytes a (possibly broken) base64 string holds; the rest stays fog. */
export function fogFromBase64(s: string): Uint8Array {
  const bits = emptyFog();
  let raw = "";
  try {
    raw = atob(s);
  } catch {
    return bits;
  }
  for (let i = 0; i < Math.min(raw.length, BYTES); i++) bits[i] = raw.charCodeAt(i);
  return bits;
}

/** As Postgres writes a bit(1024): bit 0 first. */
export function fogToBitString(bits: Uint8Array): string {
  let s = "";
  for (let i = 0; i < FOG_BITS; i++) s += hasBit(bits, i) ? "1" : "0";
  return s;
}

export function fogFromBitString(s: string): Uint8Array {
  const bits = emptyFog();
  for (let i = 0; i < Math.min(s.length, FOG_BITS); i++) if (s[i] === "1") setBit(bits, i);
  return bits;
}
```

- [ ] **Step 2: `src/server/fog.ts`**

```ts
import { emptyFog, hasBit, setBit } from "@/lib/online/fog";
import { cellBit, chunkKeyOf, type Cell } from "@/lib/world";

/**
 * The fog of war as this run of the server has seen it: which galleries it has already announced (so a visitor pacing
 * one gallery is announced once), and which bits still wait to be ORed into Postgres. A gallery explored before a
 * restart is announced again on its next visit; ORing it in again changes nothing.
 */
export class FogBook {
  private readonly seen = new Map<string, Uint8Array>();
  private readonly pending = new Map<string, Uint8Array>();

  mark(lang: string, cell: Cell): { key: string; bit: number } | null {
    const key = chunkKeyOf(cell);
    const bit = cellBit(cell);
    const id = `${lang} ${key}`;
    let seen = this.seen.get(id);
    if (!seen) this.seen.set(id, (seen = emptyFog()));
    if (hasBit(seen, bit)) return null;
    setBit(seen, bit);
    let pending = this.pending.get(id);
    if (!pending) this.pending.set(id, (pending = emptyFog()));
    setBit(pending, bit);
    return { key, bit };
  }

  takePending(): { lang: string; chunk: string; bits: Uint8Array }[] {
    const out = [...this.pending].map(([id, bits]) => {
      const space = id.indexOf(" ");
      return { lang: id.slice(0, space), chunk: id.slice(space + 1), bits };
    });
    this.pending.clear();
    return out;
  }
}
```

```ts
/** The explored galleries of some chunks of one Library (chunks nobody has been to are missing). */
export async function loadFog(lang: string, chunks: string[]): Promise<Map<string, Uint8Array>> {
  if (chunks.length === 0) return new Map();
  const rows = await sql<{ chunk: string; bits: string }[]>`
    select chunk, bits::text as bits from fog where lang = ${lang} and chunk in ${sql(chunks)}`;
  return new Map(rows.map((r) => [r.chunk, fogFromBitString(r.bits)]));
}

/** ORs explored galleries into a chunk. */
export async function addFog(lang: string, chunk: string, bits: Uint8Array): Promise<void> {
  await sql`
    insert into fog (lang, chunk, bits) values (${lang}, ${chunk}, ${fogToBitString(bits)}::bit(1024))
    on conflict (lang, chunk) do update set bits = fog.bits | excluded.bits`;
}
```

(`bit(1024)`, not `bytea`: Postgres can OR bit strings, not byte strings.)

```ts
import { cellKey, chunkOf, type Cell } from "@/lib/world";

/**
 * What the map shows: the point at the canvas centre — a gallery plus a fraction of one east (fx) and up (ff), both in
 * [0, 1) — and how many pixels a gallery takes along the row. A gallery occupies [x, x + 1) × [floor, floor + 1).
 */
export interface MapView {
  centre: Cell;
  fx: number;
  ff: number;
  scale: number;
}

/** A floor is drawn this much as tall as a gallery is wide (3.7 m against 9.7 m, a little taller to stay legible). */
export const FLOOR_RATIO = 0.55;
/** Pixels per gallery: wider than 8 would ask for too many chunks; closer than 48 shows too little. */
export const ZOOM = { min: 8, max: 48 } as const;
/** Galleries further than this from the centre of the view are off any screen. */
const FAR = 1_000_000n;

export const viewAround = (cell: Cell, scale: number, fx = 0.5, ff = 0.5): MapView => ({ centre: cell, fx, ff, scale });

/** Moves whole galleries out of the fractions into the centre. */
function normalise(view: MapView): MapView {
  const dx = Math.floor(view.fx);
  const df = Math.floor(view.ff);
  if (dx === 0 && df === 0) return view;
  return { ...view, centre: { x: view.centre.x + BigInt(dx), floor: view.centre.floor + BigInt(df) }, fx: view.fx - dx, ff: view.ff - df };
}

/** The pixel of a point of a gallery (fx, ff within it; its middle by default); Infinity when off any screen. */
export function pointOf(view: MapView, w: number, h: number, cell: Cell, fx = 0.5, ff = 0.5): [number, number] {
  const dx = cell.x - view.centre.x;
  const df = cell.floor - view.centre.floor;
  if (dx > FAR || dx < -FAR || df > FAR || df < -FAR) return [Infinity, Infinity];
  return [w / 2 + (Number(dx) + fx - view.fx) * view.scale, h / 2 - (Number(df) + ff - view.ff) * view.scale * FLOOR_RATIO];
}

/** The gallery under a pixel. */
export function cellAtPoint(view: MapView, w: number, h: number, px: number, py: number): Cell {
  const ux = view.fx + (px - w / 2) / view.scale;
  const uf = view.ff + (h / 2 - py) / (view.scale * FLOOR_RATIO);
  return { x: view.centre.x + BigInt(Math.floor(ux)), floor: view.centre.floor + BigInt(Math.floor(uf)) };
}

/** The view after dragging the map by (dx, dy) pixels. */
export const pan = (view: MapView, dx: number, dy: number): MapView =>
  normalise({ ...view, fx: view.fx - dx / view.scale, ff: view.ff + dy / (view.scale * FLOOR_RATIO) });

/** The view zoomed by `factor` (clamped to ZOOM) with the point under pixel (px, py) kept there. */
export function zoomAt(view: MapView, factor: number, px: number, py: number, w: number, h: number): MapView {
  const scale = Math.min(ZOOM.max, Math.max(ZOOM.min, view.scale * factor));
  const keep = 1 - view.scale / scale;
  const ux = (px - w / 2) / view.scale;
  const uf = (h / 2 - py) / (view.scale * FLOOR_RATIO);
  return normalise({ ...view, scale, fx: view.fx + ux * keep, ff: view.ff + uf * keep });
}

/** Keys of the chunks the canvas shows, nearest rows first. */
export function visibleChunks(view: MapView, w: number, h: number): string[] {
  const from = chunkOf(cellAtPoint(view, w, h, 0, h));
  const to = chunkOf(cellAtPoint(view, w, h, w, 0));
  const keys: string[] = [];
  for (let f = from.floor; f <= to.floor; f++) for (let x = from.x; x <= to.x; x++) keys.push(cellKey({ x, floor: f }));
  return keys;
}
```

```ts
}
```
